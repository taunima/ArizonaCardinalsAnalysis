Arizona Cardinals Season Analysis

import polars as pl
from lets_plot import *

LetsPlot.setup_html(isolated_frame=True)
card_seasons = pl.read_csv("cardinals_seasons.csv")
modern_seasons = card_seasons.filter(pl.col("season") >= 1978)

best_seasons = modern_seasons.sort("win_pct", descending=True).head(5)

era_summary = (
    card_seasons
    .group_by("era")
    .agg(
        pl.col("win_pct").mean().alias("avg_win_pct"),
        pl.col("season").count().alias("num_seasons"),
        pl.col("made_playoffs").sum().alias("playoff_appearances"),
    )
    .sort("avg_win_pct", descending=True)
)
